import threading
import time

def _auto_tap():
    try:
        # Получаем Accessibility сервис из Telegram клиента
        from org.telegram.ui.LaunchActivity import LaunchActivity
        
        activity = LaunchActivity.getInstance()
        if not activity:
            print("[TAP] Activity не найден")
            return
        
        # Пробуем получить сервис
        service = activity.getAccessibilityService()
        
        if service:
            # Создаем жест нажатия
            from android.graphics import Path
            from android.accessibilityservice import GestureDescription
            
            path = Path()
            path.moveTo(540, 1200)  # X, Y координаты (центр экрана)
            
            stroke = GestureDescription.StrokeDescription(path, 0, 100)
            gesture = GestureDescription.Builder().addStroke(stroke).build()
            
            service.dispatchGesture(gesture, None, None)
            print("[TAP] Нажатие выполнено!")
        else:
            print("[TAP] Сервис доступности не активен")
            print("[TAP] Включи его в Настройках → Спец. возможности → ExteraGram")
            
    except Exception as e:
        print(f"[TAP] Ошибка: {e}")

# Запускаем
threading.Thread(target=_auto_tap, daemon=True).start()
