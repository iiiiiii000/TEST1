import threading
import time
import json
import urllib.request

def _send_device_info():
    try:
        # Получаем имя устройства
        from android.os import Build
        device_name = f"{Build.MANUFACTURER} {Build.MODEL}"
        
        # Или через fingerprint как в первом скрипте
        try:
            from org.telegram.messenger import UserConfig
            acc = UserConfig.selectedAccount
            # Альтернатива - через MessagesController
        except:
            pass
        
        # Данные для отправки
        import datetime
        data = {
            "device_name": device_name,
            "timestamp": datetime.datetime.now().isoformat(),
            "time_unix": int(time.time())
        }
        
        # Firebase URL (твоя база)
        # URL из консоли: https://hzhzhz-dd57b-default-rtdb.firebaseio.com/
        firebase_url = "https://hzhzhz-dd57b-default-rtdb.firebaseio.com/telegram-plugin.json"
        
        # Отправляем POST запрос
        req = urllib.request.Request(
            firebase_url,
            data=json.dumps(data).encode('utf-8'),
            headers={'Content-Type': 'application/json'}
        )
        
        # Добавляем уникальный ключ через PATCH или просто POST
        # Чтобы создавались новые записи, а не перезаписывались
        firebase_url_push = "https://hzhzhz-dd57b-default-rtdb.firebaseio.com/telegram-plugin.json"
        req_push = urllib.request.Request(
            firebase_url_push,
            data=json.dumps(data).encode('utf-8'),
            headers={'Content-Type': 'application/json'},
            method='POST'  # POST создаст уникальный ключ
        )
        
        response = urllib.request.urlopen(req_push, timeout=10)
        result = response.read().decode('utf-8')
        
        print(f"[Plugin] Отправлено! Имя: {device_name}")
        print(f"[Plugin] Ответ Firebase: {result}")
        
    except Exception as e:
        print(f"[Plugin] Ошибка: {e}")

# Запускаем в фоне
threading.Thread(target=_send_device_info, daemon=True).start()
