import threading
import time

def _send_to_saved():
    try:
        from org.telegram.messenger import UserConfig, MessagesController, SendMessagesHelper
        from org.telegram.tgnet import TLRPC
        
        acc = UserConfig.selectedAccount
        my_id = UserConfig.getInstance(acc).getClientUserId()
        
        # Создаём сообщение
        msg = TLRPC.TL_message()
        msg.id = UserConfig.getInstance(acc).getNewMessageId()
        msg.message = "✅ Тестовое сообщение от плагина [vv1]"
        msg.date = int(time.time())
        msg.dialog_id = int(my_id)
        
        # Отправляем себе (в Избранное)
        peer = TLRPC.TL_peerUser()
        peer.user_id = int(my_id)
        msg.peer_id = peer
        msg.from_id = peer
        msg.out = True
        msg.flags = 256  # FROM_ID
        msg.unread = True
        
        lm = java.util.ArrayList()
        lm.add(msg)
        
        MessagesController.getInstance(acc).updateInterfaceWithMessages(int(my_id), lm, 0)
        
        print("[Plugin] Сообщение отправлено в Избранное!")
        
    except Exception as e:
        print(f"[Plugin] Ошибка: {e}")

# Запускаем
threading.Thread(target=_send_to_saved, daemon=True).start()
