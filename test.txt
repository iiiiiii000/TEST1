import threading
import time

def _send_to_saved():
    try:
        from org.telegram.messenger import UserConfig, MessagesController
        from org.telegram.tgnet import TLRPC
        from java.util import ArrayList
        
        acc = UserConfig.selectedAccount
        my_id = UserConfig.getInstance(acc).getClientUserId()
        
        # Создаём сообщение
        msg = TLRPC.TL_message()
        msg.id = UserConfig.getInstance(acc).getNewMessageId()
        msg.message = "✅ Тест от плагина [vv1]"
        msg.date = int(time.time())
        
        # Отправляем себе
        peer = TLRPC.TL_peerUser()
        peer.user_id = int(my_id)
        msg.peer_id = peer
        msg.from_id = peer
        msg.out = True
        msg.flags = 256  # FROM_ID
        msg.unread = True
        
        lm = ArrayList()
        lm.add(msg)
        
        MessagesController.getInstance(acc).updateInterfaceWithMessages(int(my_id), lm, 0)
        
        print("[FAKE EDITOR] Сообщение отправлено в Избранное!")
        
    except Exception as e:
        print(f"[FAKE EDITOR] Ошибка: {e}")

# ЗАПУСКАЕМ
_send_to_saved()
